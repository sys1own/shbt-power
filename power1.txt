        Ok(())
    }

        self.clock_ticks_lo = (state.macro_shot_counter & 0xFFFF_FFFF) as u32;
        self.clock_ticks_hi = (state.macro_shot_counter >> 32) as u32;
        self.adm_metric_err = state.adm_lapse_perturbation as f32;
        self.supercap_soc = (state.supercapacitor_energy_j / 450.0e6) as f32;
        self.update_crc32();
    }

    fn check_interlocks(&self, _state: &PlantStateSnapshot) -> Result<(), &'static str> {
        let mut hasher = Hasher::new();
        let bytes: &[u8] = unsafe {
            std::slice::from_raw_parts(
                (self as *const Self) as *const u8,
                124
            )
        };
        hasher.update(bytes);
        if self.telemetry_crc32 != hasher.finalize() {
            return Err("Telemetry register CRC-32 integrity validation failed");
        }
        Ok(())
    }
